<script>

  // Begrenzung auf Bayern

  const bayernBounds = [

    [8.97, 47.27],  // Südwest

    [13.84, 50.57]  // Nordost

  ];


  const map = new maplibregl.Map({

    container: 'map',

    style: 'https://demotiles.maplibre.org/style.json',

    center: [11.5, 49.0],

    zoom: 7.5,

    minZoom: 6,

    maxZoom: 12,           // Entspricht Ihrer maximalen Zoomstufe aus QGIS

    maxBounds: bayernBounds

  });


  map.addControl(new maplibregl.NavigationControl(), 'top-right');


  map.on('load', () => {

    map.addSource('bayern-historic', {

      type: 'raster',

      tiles: ['./tiles/{z}/{x}/{y}.png'], // Relativer Pfad zum tiles-Ordner

      tileSize: 256,

      minzoom: 6,

      maxzoom: 12,

      bounds: [8.97, 47.27, 13.84, 50.57]

    });


    map.addLayer({

      id: 'bayern-layer',

      type: 'raster',

      source: 'bayern-historic',

      paint: {

        'raster-opacity': 0.85

      }

    });


    // Deckkraft über Schieberegler steuern

    const slider = document.getElementById('slider');

    slider.addEventListener('input', (e) => {

      map.setPaintProperty('bayern-layer', 'raster-opacity', parseInt(e.target.value, 10) / 100);

    });

  });

</script>
